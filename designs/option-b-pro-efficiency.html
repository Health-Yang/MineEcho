Here's a comprehensive script with pandas and matplotlib. It handles date parsing, monthly aggregation, and generates both a bar chart and trend line.

sales_analysis.py 
import pandas as pd
import matplotlib.pyplot as plt
from datetime import datetime

def analyze_sales(filepath: str) -> pd.DataFrame:
    """Analyze sales CSV and return monthly aggregates."""
    df = pd.read_csv(filepath)
    df['date'] = pd.to_datetime(df['date'])
    df['month'] = df['date'].dt.to_period('M')

    monthly = df.groupby('month').agg({
        'amount': ['sum', 'mean', 'count'],
        'quantity': 'sum'
    }).reset_index()

    return monthly

if __name__ == "__main__":
    results = analyze_sales("sales.csv")
    print(f"Processed {len(results)} months")
    results.to_csv("monthly_summary.csv", index=False)
