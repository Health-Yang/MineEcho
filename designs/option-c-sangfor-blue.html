import pandas as pd
import matplotlib.pyplot as plt

def analyze_sales(csv_path):
    df = pd.read_csv(csv_path)
    df['date'] = pd.to_datetime(df['date'])
    monthly = df.groupby(df['date'].dt.to_period('M'))['amount'].sum()
    return monthly
